				transfer_state_to = SOV
				add_core_of = SOV
			}
		}
			
	}
	
	option = {
		name = crimea.15.b

				puppet = CRI
			}

			}
	}
	
	option = {
		name = crimea.15.c
		
		controller = { 
			SOV = {
				create_wargoal = {
				target = CRI
				type = take_state_focus

				}
				
				add_ai_strategy = {
					type = declare_war
					id = CRI
					value = 100
				}
			}
			
			GER = {
				add_ai_strategy = {
					type = dont_join_wars_with

				}
			}
		}
